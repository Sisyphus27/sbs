**Goal:** Add GZCLP T2 (back) and T3 (accessories) progression tiers to `SBS RTF filled GZCLP.xlsx`, leaving the SBS T1/auxiliary engine untouched.

**Architecture:** A Python/openpyxl tool under `tools/sbs_gzclp/` builds the new tiers. Pure-Python functions (`progression.py`) encode the T2 state machine and T3 rule as the spec source-of-truth; formula generators (`formulas.py`) emit the exact Excel strings that mirror those functions; a builder (`builder.py`) injects Quick Setup config + T2/T3 zones into each day of each frequency sheet. Tests (`pytest`) drive TDD on the pure logic and verify generated formula strings; integration tests run on temp copies of the real workbook.

**Tech Stack:** Python 3 (conda env `tamp`), openpyxl, pytest. Excel formulas use `MROUND` + `IF`; rounding reuses `'Quick Setup'!$A$2`.

**Spec:** `docs/superpowers/specs/2026-06-25-sbs-gzclp-t2-t3-design.md`

```python
"""Weekly block column arithmetic for the SBS day sheets.

Each week occupies a 7-column block. Week-1 block starts at column B (index 2).
Block starts: B, I, P, W, AD, AK, ... (step 7), 21 weeks total.
Within a block (0-based offset from the block start):
    0 = Weight          (B, I, P, ...)
    1 = Target reps     (C, J, Q, ...)
    2 = FailStreak T2   (D, K, R, ...)   [T2 state row only]
    3 = (unused / Set goal)
    4 = Reps on last set (ACTUAL log) (F, M, T, ...)
    5 = Video
    6 = Notes
"""
from openpyxl.utils import get_column_letter
from openpyxl.utils.cell import column_index_from_string


def block_start_cols(weeks: int = 21, start: int = 2, step: int = 7) -> list[str]:
    """Return the column letters of each weekly block's start column."""
    return [get_column_letter(start + i * step) for i in range(weeks)]


def col(block_start: str, offset: int) -> str:
    """Return the column letter at `offset` columns right of `block_start`."""
    return get_column_letter(column_index_from_string(block_start) + offset)


BLOCKS: list[str] = block_start_cols(21, 2, 7)
```

`tools/sbs_gzclp/progression.py`:
```python
"""Pure-Python encoding of the GZCLP T2 state machine and T3 rule.

These functions are the source of truth. The Excel formula generators in
formulas.py MUST mirror them. Mirroring is enforced by test_formulas.py.
"""
import math
from dataclasses import dataclass
from typing import Optional


@dataclass(frozen=True)
class T2Params:
    incr: float      # weekly increment (kg), e.g. 2.5
    reset: float     # deload multiplier on full reset, e.g. 0.8
    fail: int        # consecutive misses to trigger a tier change, e.g. 3


@dataclass(frozen=True)
class T3Params:
    target: int      # last-set target reps, e.g. 15
    incr: float      # weekly increment (kg), e.g. 2.5


def t2_next(target: int, streak: int, weight: float,
            last_set: Optional[int], p: T2Params) -> tuple[int, int, float]:
    """Return (next_target, next_streak, next_weight) for T2 (Back).

    Scheme tiers: 10 -> 8 -> 6. On the 3rd consecutive miss at a tier, drop a
    tier (same weight); at 6, the 3rd miss is a full reset (weight * reset,
    back to 10). A hit adds `incr` and keeps the current tier. No log yet
    carries state forward unchanged.
    """
    if last_set is None:
        return target, streak, weight

    hit = last_set >= target
    if hit:
        return target, 0, round_weight(weight + p.incr)

    # missed
    if streak + 1 >= p.fail:
        # tier change triggered
        if target == 10:
            return 8, 0, weight
        if target == 8:
            return 6, 0, weight
        # target == 6: full reset
        return 10, 0, round_weight(weight * p.reset)

    # miss but under threshold: accumulate streak, same weight/scheme
    return target, streak + 1, weight


def t3_next(weight: float, last_set: Optional[int], p: T3Params) -> float:
    """Return next weight for T3 (Accessories): +incr if last_set >= target."""
    if last_set is None:
        return weight
    if last_set >= p.target:
        return round_weight(weight + p.incr)
    return weight


def round_weight(w: float, quantum: float = 2.5) -> float:
    """Mirror Excel MROUND(w, quantum): round w/quantum half-away-from-zero, then * quantum."""
    n = math.floor((w / quantum) + 0.5)   # positive weights only; half away from zero
    return round(n * quantum, 10)
```

```python
"""Excel formula-string generators for T2/T3, mirroring progression.py.

Column layout within a weekly block (offset from block start):
    0 = Weight, 1 = Target, 2 = FailStreak (T2 state row), 4 = Reps on last set.

Quick Setup param cells (absolute refs, robust vs named-range API):
    T2_incr  = 'Quick Setup'!$B$71
    T2_reset = 'Quick Setup'!$B$72
    T2_fail  = 'Quick Setup'!$B$73
    T3_target= 'Quick Setup'!$B$74
    T3_incr  = 'Quick Setup'!$B$75
Rounding quantum: 'Quick Setup'!$A$2
"""
from tools.sbs_gzclp.columns import col

QS = "'Quick Setup'"
T2_INCR = f"{QS}!$B$71"
T2_RESET = f"{QS}!$B$72"
T2_FAIL = f"{QS}!$B$73"
T3_TARGET = f"{QS}!$B$74"
T3_INCR = f"{QS}!$B$75"
QUANTUM = f"{QS}!$A$2"


def t2_target_formula(prev_block, next_block, state_row, log_row) -> str:
    pc = lambda o: f"{col(prev_block, o)}{state_row}"      # prev state cells
    pl = f"{col(prev_block, 4)}{log_row}"                  # prev last-set (log row)
    target, streak = pc(1), pc(2)
    return (f"=IF({pl}=\"\",{target},"
            f"IF({pl}>={target},{target},"
            f"IF({streak}+1>={T2_FAIL},IF({target}=10,8,IF({target}=8,6,10)),{target})))")


def t2_streak_formula(prev_block, next_block, state_row, log_row) -> str:
    pc = lambda o: f"{col(prev_block, o)}{state_row}"
    pl = f"{col(prev_block, 4)}{log_row}"
    target, streak = pc(1), pc(2)
    return (f"=IF({pl}=\"\",{streak},"
            f"IF({pl}>={target},0,"
            f"IF({streak}+1>={T2_FAIL},0,{streak}+1)))")


def t2_weight_formula(prev_block, next_block, state_row, log_row) -> str:
    pc = lambda o: f"{col(prev_block, o)}{state_row}"
    pl = f"{col(prev_block, 4)}{log_row}"
    weight, target, streak = pc(0), pc(1), pc(2)
    return (f"=IF({pl}=\"\",{weight},"
            f"IF({pl}>={target},MROUND({weight}+{T2_INCR},{QUANTUM}),"
            f"IF(AND({streak}+1>={T2_FAIL},{target}=6),"
            f"MROUND({weight}*{T2_RESET},{QUANTUM}),{weight})))")


def t3_weight_formula(prev_block, next_block, row) -> str:
    weight = f"{col(prev_block, 0)}{row}"
    last = f"{col(prev_block, 4)}{row}"
    return (f"=IF({last}=\"\",{weight},"
            f"IF({last}>={T3_TARGET},MROUND({weight}+{T3_INCR},{QUANTUM}),{weight}))")


def t3_target_formula(row) -> str:
    return f"={T3_TARGET}"
```

```python
from tools.sbs_gzclp.columns import BLOCKS, col
from tools.sbs_gzclp import formulas as F

DAY_SHEETS = ["2x", "3x", "4x", "5x", "6x"]
ACCESSORIES_PER_DAY = 3   # default T3 rows per day (user can extend slot table)


def _accessories_label_rows(ws) -> list[int]:
    return [r for r in range(1, ws.max_row + 1)
            if ws.cell(row=r, column=1).value == "Accessories"]


def inject_t3_zones(path: str) -> None:
    wb = openpyxl.load_workbook(path)
    for sh in DAY_SHEETS:
        ws = wb[sh]
        label_rows = _accessories_label_rows(ws)
        for day_idx, label_row in enumerate(label_rows):
            for k in range(ACCESSORIES_PER_DAY):
                data_row = label_row + 1 + k
                slot_row = T3_FIRST_DATA_ROW + day_idx * ACCESSORIES_PER_DAY + k
                _write_t3_row(ws, data_row, slot_row)
    wb.save(path)


def _write_t3_row(ws, row: int, slot_row: int) -> None:
    """Write one T3 accessory lift into `row`, wired to Quick Setup T3 slot `slot_row`."""
    # Name
    ws.cell(row=row, column=1).value = f"='Quick Setup'!A{slot_row}"
    # Week 1 (block B): weight from slot start, target = T3_target
    ws[f"B{row}"].value = f"='Quick Setup'!B{slot_row}"
    ws[f"C{row}"].value = F.t3_target_formula(row)
    # Weeks 2..21: weight progression referencing previous block
    for i in range(1, len(BLOCKS)):
        prev_block, next_block = BLOCKS[i - 1], BLOCKS[i]
        ws[f"{col(next_block, 0)}{row}"].value = F.t3_weight_formula(prev_block, next_block, row)
        # keep target column showing the param too (optional, helps readability)
        ws[f"{col(next_block, 1)}{row}"].value = F.t3_target_formula(row)
```

```python
BACK_PER_DAY = 2
T2_ROWS_PER_LIFT = 2   # state row + log row


def inject_t2_zones(path: str) -> None:
    wb = openpyxl.load_workbook(path)
    for sh in DAY_SHEETS:
        ws = wb[sh]
        label_rows = _accessories_label_rows(ws)
        # Process days bottom-up so insertions in one day don't shift later days' anchors.
        for day_idx in range(len(label_rows) - 1, -1, -1):
            label_row = label_rows[day_idx]
            needed = BACK_PER_DAY * T2_ROWS_PER_LIFT
            gap = _empty_rows_above(ws, label_row)
            if gap < needed:
                ws.insert_rows(label_row, needed - gap)
                # recompute label rows after insert on this sheet
                label_rows = _accessories_label_rows(ws)
                label_row = label_rows[day_idx]
            first_state_row = label_row - needed
            for k in range(BACK_PER_DAY):
                state_row = first_state_row + k * T2_ROWS_PER_LIFT
                log_row = state_row + 1
                slot_row = T2_FIRST_DATA_ROW + day_idx * BACK_PER_DAY + k
                _write_t2_lift(ws, state_row, log_row, slot_row)
    wb.save(path)


def _empty_rows_above(ws, row: int, ncols: int = 30) -> int:
    gap = 0
    rr = row - 1
    while rr > 0 and all(ws.cell(row=rr, column=c).value is None for c in range(1, ncols)):
        gap += 1; rr -= 1
    return gap


def _write_t2_lift(ws, state_row: int, log_row: int, slot_row: int) -> None:
    # Name on state row
    ws.cell(row=state_row, column=1).value = f"='Quick Setup'!A{slot_row}"
    # Week 1 seed (block B)
    ws[f"B{state_row}"].value = f"='Quick Setup'!B{slot_row}"   # weight = slot start
    ws[f"C{state_row}"].value = 10                              # target
    ws[f"D{state_row}"].value = 0                               # fail streak
    # log row week-1 last-set (F) intentionally blank
    # Weeks 2..21: state formulas reference previous block
    for i in range(1, len(BLOCKS)):
        prev_block, next_block = BLOCKS[i - 1], BLOCKS[i]
        ws[f"{col(next_block, 1)}{state_row}"].value = F.t2_target_formula(prev_block, next_block, state_row, log_row)
        ws[f"{col(next_block, 2)}{state_row}"].value = F.t2_streak_formula(prev_block, next_block, state_row, log_row)
        ws[f"{col(next_block, 0)}{state_row}"].value = F.t2_weight_formula(prev_block, next_block, state_row, log_row)
```
