```python
def t2_next(state: T2State, actual, est1rm: float, fail: int = 3,
            incr: float = 2.5, reset_pct: float = 0.70, quantum: float = 2.5) -> T2State:
    """GZCLP-modified T2 back: 4x8 -> 4x6 cascade; reset = reset_pct * est1rm, back to 8."""
    if actual is None:
        return state
    if actual >= state.target:                                   # hit
        return T2State(state.target, 0, round_weight(state.weight + incr, quantum))
    if state.streak + 1 >= fail:                                 # Nth consecutive miss
        if state.target == 8:
            return T2State(6, 0, state.weight)                   # 4x8 -> 4x6
        return T2State(8, 0, round_weight(est1rm * reset_pct, quantum))  # at 6 -> reset to 8
    return T2State(state.target, state.streak + 1, state.weight) # miss, under threshold
```

- [ ] **Step 1: Write the one-shot migration script**

Create `scripts/migrate_t2_4x8.py`:

```python
"""One-shot: migrate live sbs.db to T2 4x8 + swap Barbell rows <-> DB rows days.
Run once, then delete. Backs up first."""
import shutil
from webapp import db, repo

DB = "sbs.db"
shutil.copy(DB, "backups/sbs-t2-4x8.db.bak")
c = db.connect(DB)
db.init_schema(c)

# 1. All T2 lifts: sets 3 -> 4, lift_state.target 10 -> 8 (weight/state/history unchanged).
for r in repo.list_lifts(c):
    if r["tier"] != "t2":
        continue
    repo.update_lift(c, r["id"], sets=4)
    st = repo.get_lift_state(c, r["id"])
    repo.save_lift_state(c, r["id"], tier="t2", tm=st["tm"], weight=st["weight"],
                         target=8, streak=st["streak"], est1rm=st["est1rm"])

# 2. Back-lift day+sort_order swap (weights follow each lift).
a = repo.get_lift_by_name(c, "Barbell rows")
b = repo.get_lift_by_name(c, "DB rows")
repo.update_lift(c, a["id"], day=b["day"], sort_order=b["sort_order"])
repo.update_lift(c, b["id"], day=a["day"], sort_order=a["sort_order"])

print("--- T2 lifts after migration ---")
for r in repo.list_lifts(c):
    if r["tier"] == "t2":
        st = repo.get_lift_state(c, r["id"])
        print(f"  {r['name']}: day={r['day']} sets={r['sets']} target={st['target']} weight={st['weight']}")
c.close()
```
